Запуск программы:
julia Main.jl <имя конфига>

Отрисовка графиков:
julia MakeFigures.jl <имя конфига>


Пример Config-файла:

# --- параметры расчетной сетки ---
L    = 2           # длина расчетной области
N    = 101         # количество рельных узлов 
fict = 3           # количество фиктивных узлов 
h    = L / (N - 1) # постоянный шаг сетки

####################################################################################################

# --- физические параметры задачи ---
b = 50 # крутизна границы раздела плотностей 

rho1A = 90   # равновесная плотность вещества 1 в фазе А (жидкой)
rho1B = 1200 # равновесная плотность вещества 1 в фазе В (твердой)
rho2A = 900  # равновесная плотность вещества 2 в фазе А (жидкой)
rho2B = 120  # равновесная плотность вещества 2 в фазе В (твердой)

lam11 = 4e-8 # капиллярный коэффициент в выражении для энергии коверхностного натяжения
lam12 = 0
lam21 = lam12
lam22 = lam11

Apsi = 1e-4 # коэффициент жесткости фазы А 
Bpsi = Apsi # коэффициент жесткости фазы А 

lamel = 2e2   # первый коэффициент Ламе
muel  = lamel # второй коэффициент Ламе

theta = 1 # абсолютная температура

M0 = 1 # мобильность

eta  = 5 # динамическая вязкость
zeta = 0 # объемная вязкость

g = 0.05 # ускорение свободного падения

####################################################################################################

# --- начальные условия ---
u_init   = -0.1 # начальная скорость всей системы
rho_init = "smooth" # начальная плотность: "smooth" - плавная граница раздела фаз,
                    #                      "sharp" - резкая граница раздела фаз, 
                    #                      "spinodal" - неустойчивая смесь для спинодального распада

# --- граничные условия ---
bound = "wall" # "wall" - жесткая стенка,
               # "periodic" - периодическое условие

####################################################################################################

# --- параметры времени ---
t_max    = 1.0           # максимальное время 
delta_t  = 1e-6          # шаг по времени
step_max = t_max/delta_t # количество шагов
frame_dt = 0.1           # как часто сохранять результаты расчетов

####################################################################################################

# --- параметры поршня ---
piston_width = 0.3 * L # ширина поршня
x0_piston    = 0.6 * L # начальное положение центра масс поршня
v0_piston    = u_init  # начальная скорость центра масс поршня
#S            = 1.0
#gamma        = 1.0

####################################################################################################

# --- директория для сохранения картинок, лог-файла и конфиг-файла ---
direct = joinpath("Figures", replace(basename(@__FILE__), ".jl" => ""))
if !isdir(direct)
    mkpath(direct)
end
cp(config_file, joinpath(direct, config_file), force=true)

####################################################################################################

# --- подобранные вручную коэффициенты масштабирования графиков ---
down = -0.11 # на сколько опущены графики

rho_max_coeff = 1.75 # коэффициент для расчета примерной максимальной плотности

rho_min = -100 # минимальная плотность

u_max_coeff = 0.2 # коэффициент для расчета примерной максимальной скорости
u_min_coeff = 2.0 # коэффициент для расчета примерной минимальной скорости

mu_max_coeff = 1.4 # коэффициент для расчета примерного максимального химического потенциала
mu_min_coeff = 1.1 # коэффициент для расчета примерного максимального химического потенциала

p_max_coeff = 1.2 # коэффициент для расчета примерного максимального давления
p_min_coeff = 0.6 # коэффициент для расчета примерного минимального давления

E_max_coeff = 0.5 # коэффициент для расчета примерной максимальной полной энергии
E_min_coeff = 0.2 # коэффициент для расчета примерной минимальной полной энергии

####################################################################################################

# --- подгоночные коэффициенты приближенного аналитического решения ---
#c_coeff = 5.2
#m_coeff = 0.26

####################################################################################################

# --- точности при проверке сохранения массы и полной энергии ---
mass_rtol   = 1e-8
energy_rtol = 1e-6

####################################################################################################

# список величин для построения отдельных графиков и сабплотнов на одном графике ---
requests = [:density, :energy, :mu, [:density, :energy, :velocity, :mu]]
suffix = Dict(:density => "rho", :energy => "E", :velocity => "u", :pressure => "p", :mu => "mu")
for sub in unique(vcat(collect(values(suffix)), "mp4"))
    mkpath(joinpath(direct, sub))
end
